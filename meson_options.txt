# Real PAM auth for the native lockscreen. Disable to build the headless test
# stub (a fixed password), so CI/soak links without libpam or a real login.
option('pam', type: 'feature', value: 'enabled',
  description: 'PAM authentication for the native lockscreen')

# The shared DBus connection (notifications, systray). sd-bus, from libsystemd
# or basu; disable to build with no bus at all, so a headless CI box needs no
# session bus. Disabled means ABSENT, not stubbed: no name is owned, no notify
# or tray event ever arrives, and the verbs are the no-op an unset op table
# already is.
option('dbus', type: 'feature', value: 'enabled',
  description: 'DBus services (notifications, systray)')

# XWayland (X11 clients). Enabled requires wlroots with xwayland and xcb for
# the WM-side connection; disable to build without any X11 surface, so
# headless CI needs no Xwayland on the machine.
option('xwayland', type: 'feature', value: 'enabled',
  description: 'XWayland support (X11 clients)')
